// apps/frontend/src/lib/api/utils/auth/decryptAndVerifyIdToken.ts
import * as jose from 'jose';
import { constants } from '$lib/server/constants';
import { constants as publicConstants } from '$lib/utils/constants';

export interface DecryptAndVerifyOptions {
  privateEncryptionJWKSet: Array<jose.JWK>;
  publicSignatureJWKSetUri: string;
  audience?: string;
  issuer?: string;
}

export const defaultOptions: DecryptAndVerifyOptions = {
  /**
   * Parsed lazily, ON READ, rather than at module-evaluation time. [... D-22: the whole
   * getIdTokenClaims.ts:6-15 block moves here VERBATIM, unrewritten ...]
   *
   * Do not simplify this back into a plain property: that reinstates an
   * uncatchable import-time crash.
   */
  get privateEncryptionJWKSet(): Array<jose.JWK> {
    try {
      return JSON.parse(constants.IDENTITY_PROVIDER_DECRYPTION_JWKS || '[]');
    } catch {
      // Opaque identifier only -- the malformed value is NOT echoed.
      throw Object.assign(new Error('Cannot decode ID token: the configured decryption JWK set is not valid JSON.'), {
        code: 'ERR_JWKS_MALFORMED'
      });
    }
  },
  /** D-07: a getter, not a property. Do not simplify this back into a plain property: that
   *  converts a per-call read into an import-time snapshot and breaks the beforeAll seam. */
  get publicSignatureJWKSetUri(): string {
    return constants.IDENTITY_PROVIDER_JWKS_URI!;
  },
  /** D-07: a getter, not a property. (same warning) */
  get audience(): string {
    return publicConstants.PUBLIC_IDENTITY_PROVIDER_CLIENT_ID;
  },
  /** D-07: a getter, not a property. (same warning) */
  get issuer(): string {
    return constants.IDENTITY_PROVIDER_ISSUER;
  }
};

export async function decryptAndVerifyIdToken(
  idToken: string,
  options: DecryptAndVerifyOptions = defaultOptions
): Promise<jose.JWTPayload> {
  const { kid } = jose.decodeProtectedHeader(idToken);

  // [... D-22: the whole getIdTokenClaims.ts:46-53 leak-safety block moves here VERBATIM ...]
  const privateEncryptionJWKSet = options.privateEncryptionJWKSet;

  if (privateEncryptionJWKSet.length === 0) {
    throw Object.assign(new Error('Cannot decode ID token: no decryption JWKs are configured.'), {
      code: 'ERR_JWKS_EMPTY'
    });
  }

  const privateEncryptionJWK = privateEncryptionJWKSet.find((jwk) => jwk.kid === kid);

  if (!privateEncryptionJWK) {
    throw Object.assign(new Error(`Cannot decode ID token: JWK not found: kid=${kid}.`), {
      code: 'ERR_JWK_KID_MISMATCH'
    });
  }

  const { plaintext } = await jose.compactDecrypt(idToken, await jose.importJWK(privateEncryptionJWK));
  const { payload } = await jose.jwtVerify(
    new TextDecoder().decode(plaintext),
    jose.createRemoteJWKSet(new URL(options.publicSignatureJWKSetUri)),
    { audience: options.audience, issuer: options.issuer }
  );
  return payload;
}

### 1.2 Design points, each with its evidence

| Point | Answer | Evidence |
|---|---|---|
| **Return type** | `Promise<jose.JWTPayload>` is correct. `jose.jwtVerify` yields `{ payload: JWTPayload }`, and **both** providers' post-verify mapping is pure index access — `String(payload[CONFIG.firstNameProp] ?? '')` etc. `JWTPayload` carries `[propName: string]: unknown`, so `payload[claim]` compiles with no cast and `String(… ?? '')` narrows it. | [VERIFIED: `idura.ts:135-146`, `signicat.ts:98-109`] + [MEASURED: 0 typecheck errors] |
| **Why `throw`, not a result union** | D-06 fixes it, and it is also what makes the D-02b repoint mechanical: each provider's *existing* catch arm (§ 2.3) already converts a `code`-carrying `Error` into `{success:false,error:{code}}`. Nothing about the catch arm changes. | [VERIFIED: `idura.ts:148-161`, `signicat.ts:111-124`] |
| **Why an explicit `DecryptAndVerifyOptions` interface, not `as const`** | The deleted helper used `} as const;` (`getIdTokenClaims.ts:29`) plus a structurally-typed inline parameter (`:33-38`). With four getters, annotating the object with a named exported interface is what lets the default-parameter type and the explicit-options call sites (§ 4) agree without a second structural literal drifting from the first. | [MEASURED: compiles 0/0] |
| **Statement order is load-bearing — keep it** | `decodeProtectedHeader` **first**, then the empty-set check, then the kid lookup. ## Common Pitfalls

### Pitfall 1 — the zero-delta injection
**What goes wrong:** D-18's `(hardcode 'birthdate')` is applied to Signicat, the run stays green, and
the executor concludes the remediation failed (or worse, records a green `NEW-assertion outcome`).
**Why:** `SIGNICAT_AUTH_CONFIG.identityMatchProp` **is** `'birthdate'`.
**Avoid:** § 6.1's per-provider literal. **Warning sign:** an injected run whose whole-file count is
identical to the clean run.

### Pitfall 2 — crediting a collateral red
**What goes wrong:** the lazy-parse injection reds three tests; the claims test's red is written into
the `NEW-assertion outcome` cell.
**Avoid:** D-08's two-column rule, pre-registered per pair in § 6.2. Only a red **naming the
strengthened assertion** may fill that cell.

### Pitfall 3 — `sub` set in `claims` instead of `opts.subject`
**What goes wrong:** Idura's `identifier` deep-equal reds against `'test-subject'` and the executor
"fixes" the assertion instead of the fixture.
**Why:** `createTestJwe` calls `.setSubject(opts?.subject ?? 'test-subject')` **after** the payload,
so it overwrites. [VERIFIED: `tokens.ts:43-50`]

### Pitfall 4 — the `signicat.ts` lint red mistaken for a collapse failure
**What goes wrong:** `yarn lint:check` reds on an unused `jose` import and the collapse is suspected.
**Avoid:** § 2.2 — delete the import and run `eslint --fix`. [MEASURED]

### Pitfall 5 — port mismatch aborting a gate before spec one
**What goes wrong:** `FRONTEND_PORT=5174` on the Playwright side, dev server on `5173`; the preflight
aborts and the failure reads like a product break.
**Avoid:** § 7.3. There is **no** bypass by design. [VERIFIED: `tests/global-setup.ts:19-22`]

### Pitfall 6 — trusting `bank-auth-journey.setup.ts`'s header over the config
**What goes wrong:** the header says the project "stands alone"; the run takes 11 minutes and looks
wrong. **Avoid:** § 7.1 — the config (`:494`) is authoritative; the comment is stale.

### Pitfall 7 — reversing the core's statement order
**What goes wrong:** putting the empty-set check before `decodeProtectedHeader` changes ten
`token-endpoint.test.ts` failures from uncoded to `ERR_JWKS_EMPTY`.
**Avoid:** § 1.2 — keep the helper's order.
