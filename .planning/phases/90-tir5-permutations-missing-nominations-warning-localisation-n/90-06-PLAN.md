    ```typescript
    /**
     * Build the post-click URL match regex for the `langSelectorFixture.switchTo`
     * Playwright fixture (tests/tests/fixtures/candidate/langSelectorFixture.fixture.ts).
     *
     * Paraglide serves the `baseLocale` from `/` (no `/<locale>/` prefix); all
     * other locales mount under `/<locale>/`. This helper handles both branches.
     *
     * Closes Phase 90 VERIFICATION Gap #2 / REVIEW CR-02 (Plan 90-06).
     *
     * IMPORTANT — this module is the SINGLE SOURCE OF TRUTH for the regex
     * contract. A near-identical implementation lives inside the Playwright
     * fixture (cannot cross-workspace import from Playwright into vitest scope
     * because tests/ has no vitest config). A CONTRACT TEST in
     * `tests/langSelectorRegex.test.ts` asserts both copies produce byte-
     * identical regex sources — any drift between this file and the fixture
     * fails CI.
     *
     * @param locale  target locale code (one of paraglideLocales).
     * @returns RegExp matching the post-switch URL shape for `locale`.
     */
    import { baseLocale, locales as paraglideLocales } from '$lib/paraglide/runtime';

    export function buildLocaleSwitchUrlRegex(locale: string): RegExp {
      if (locale === baseLocale) {
        const nonBase = paraglideLocales.filter((l) => l !== baseLocale).join('|');
        return new RegExp(`^https?://[^/]+/(?!(?:${nonBase})(?:/|$))`);
      }
      return new RegExp(`^https?://[^/]+/${locale}(?:/|$)`);
    }
    ```
